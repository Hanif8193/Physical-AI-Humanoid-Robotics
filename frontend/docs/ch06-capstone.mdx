### 6.3 The Demo Task: "Bring Me the Red Bottle"

The capstone demonstration task:
1. **Humanoid hears**: "Please bring me the red bottle from the table"
2. **Whisper transcribes** the speech to text
3. **GPT-4o plans**: NAVIGATE(table) → LOOK_AT(red bottle) → PICK(red bottle) → NAVIGATE(user) → PLACE(red bottle, user_hand)
4. **Nav2 navigates** to the table using the map built by cuVSLAM
5. **Object detector** (YOLOv8) identifies and localizes the red bottle in 3D
6. **MoveIt2** plans the arm trajectory to the bottle
7. **Gripper** grasps the bottle
8. **Nav2** navigates back to the user
9. **Humanoid speaks**: "Here is your red bottle."

### 6.4 Implementation: The Orchestrator Node

```python
#!/usr/bin/env python3
"""
Autonomous Humanoid Orchestrator
Integrates all layers: Perception → Cognition → Execution
"""
import rclpy
from rclpy.node import Node
from rclpy.callback_groups import ReentrantCallbackGroup
from std_msgs.msg import String
from vision_msgs.msg import Detection3DArray
import json
import asyncio

class AutonomousHumanoid(Node):
    """Main orchestrator for the autonomous humanoid."""

    def __init__(self):
        super().__init__('autonomous_humanoid')
        self._cb_group = ReentrantCallbackGroup()

        # Subscriptions
        self.create_subscription(
            String, '/transcription', self._on_command, 10,
            callback_group=self._cb_group
        )
        self.create_subscription(
            Detection3DArray, '/detected_objects', self._on_detection, 10,
            callback_group=self._cb_group
        )

        # Publishers
        self.action_pub = self.create_publisher(String, '/action_plan', 10)
        self.speech_pub = self.create_publisher(String, '/tts_input', 10)

        # State
        self._current_task = None
        self._detections = {}
        self._planner = self._init_planner()

        self.get_logger().info('🤖 Autonomous Humanoid ready. Listening for commands...')

    def _init_planner(self):
        from openai import OpenAI
        return OpenAI()

    def _on_command(self, msg: String):
        """Handle incoming voice command."""
        command = msg.data.strip()
        if not command or len(command) < 3:
            return

        self.get_logger().info(f'📢 Received command: "{command}"')
        self._current_task = command

        # Build context from current detections
        context = {
            "detected_objects": list(self._detections.keys()),
            "robot_location": "home_position",
        }

        # Plan with GPT-4o
        plan = self._plan_task(command, context)
        self.get_logger().info(f'📋 Plan: {json.dumps(plan, indent=2)}')

        # Execute
        self.action_pub.publish(String(data=json.dumps(plan)))

    def _plan_task(self, command: str, context: dict) -> dict:
        """Use GPT-4o to decompose task into actions."""
        response = self._planner.chat.completions.create(
            model="gpt-4o",
            messages=[
                {
                    "role": "system",
                    "content": (
                        "You are a robot task planner. Convert commands to action sequences.\n"
                        "Available actions: NAVIGATE(x,y), LOOK_AT(obj), PICK(obj), "
                        "PLACE(obj,target), SPEAK(text).\n"
                        "Return JSON: {\"actions\": [{\"action\": ..., \"args\": [...]}]}"
                    ),
                },
                {
                    "role": "user",
                    "content": f"Command: {command}\nContext: {json.dumps(context)}",
                },
            ],
            response_format={"type": "json_object"},
        )
        return json.loads(response.choices[0].message.content)

    def _on_detection(self, msg: Detection3DArray):
        """Update object detection state."""
        self._detections = {
            det.results[0].hypothesis.class_id: det.bbox.center
            for det in msg.detections
            if det.results
        }


def main():
    rclpy.init()
    node = AutonomousHumanoid()
    executor = rclpy.executors.MultiThreadedExecutor()
    executor.add_node(node)
    executor.spin()
    rclpy.shutdown()


if __name__ == '__main__':
    main()
```
